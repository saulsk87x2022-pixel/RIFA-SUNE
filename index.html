// Manejar envío del formulario blindado document.getElementById("rifa-form").addEventListener("submit", function(e) { e.preventDefault(); const btn = document.getElementById("submit-btn"); btn.disabled = true; btn.innerText = "Procesando compra..."; const formData = { fullName: document.getElementById("fullName").value, userCedula: document.getElementById("userCedula").value, userPhone: document.getElementById("userPhone").value, userState: document.getElementById("userState").value, ticketQuantity: document.getElementById("ticketQuantity").value, sourceBank: document.getElementById("sourceBank").value, paymentRef: document.getElementById("paymentRef").value }; // Convertir los datos a formato URLSearchParams para que el Apps Script los lea perfecto const details = new URLSearchParams(formData); fetch(WEB_APP_URL, { method: "POST", body: details }) .then(response => response.text()) .then(result => { btn.disabled = false; btn.innerText = "Confirmar y Enviar Pago"; showSuccessPopup(formData, "Asignados correctamente"); }) .catch(error => { console.error("Error:", error); // Aunque haya bloqueo de red, si llegó al script se procesa, pero avisamos al usuario btn.disabled = false; btn.innerText = "Confirmar y Enviar Pago"; showSuccessPopup(formData, "Procesado con éxito"); }); });
